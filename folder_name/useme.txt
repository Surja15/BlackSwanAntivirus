void scanFile(const char* filePath, YR_RULES* rules) {
    MatchList matchList = {0};

    int res = yr_rules_scan_file(
        rules, filePath,
        SCAN_FLAGS_REPORT_RULES_MATCHING,
        scanCallback,
        &matchList,
        0
    );

    if (res != ERROR_SUCCESS) {
        fprintf(stderr, "[-] Scan error: %s\n", filePath);
        return;
    }

    if (matchList.count > 0) {
        CallQuarantine(filePath, &matchList);
    }

    for (int i = 0; i < matchList.count; i++)
        free(matchList.matches[i]);
}
